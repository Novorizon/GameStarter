using NavigationLab.Core;
using NavigationLab.DAoe4;
var lines=File.ReadAllLines("/workspace/navigation-comparison/d-aoe4/Assets/GameAssets/TowerDefense/Data/ForestNavigation.bytes").Where(l=>l.Length>0&&!l.StartsWith(';')).ToArray();

var walk=new bool[width*height];var hold=new bool[width*height];var spawns=new List<int>();

var world=new WorldDefinition(width,height,cell,new Point2(origin[0],origin[1]),walk,hold,spawns.ToArray(),new[]{new TargetDefinition(1,new Point2(target[0],target[1]),target[2])});

using var nav=(args.Contains("--yield")?(INavigationFactory)new Aoe4CoordinatedFactory():new Aoe4Factory()).Create(world,new[]{small,big});

for(int tick=0;tick<500;tick++)
{
 if(next<=count&&nav.TrySpawn(new SpawnRequest(next,big,spawns[0]))){next++;}
 for(int s=0;s<10;s++){nav.Step();}
 var units=nav.Snapshot();
 if(tick%200==0){Console.WriteLine($"{tick} born {next-1} settled {units.Count(u=>u.State!=MotionState.Moving)} last={nav.LastSpawnFailure}");}
 if(next==count+1&&units.All(u=>u.State!=MotionState.Moving)){Console.WriteLine($"DONE tick{tick}");break;}
 if(tick==499){foreach(var u in units){Console.WriteLine($"id{u.Id} pos{u.Position.X},{u.Position.Y} goal{u.GoalCell} state{u.State}");}}
}
